El script debe:
· Buscar en "Cuentas" col D = nº cuenta → obtener email col A
· Buscar en "COMANDES" primera fila Available con Product+Color+Size
· Actualizar: For=email, Status=Sold, Sold=precio, Date=hoy
Ver código Apps Script completo
function doPost(e) { var d = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Buscar email en hoja Cuentas (col D = numCuenta, col A = email) var cuentasSheet = ss.getSheetByName("Cuentas"); var cuentasData = cuentasSheet.getDataRange().getValues(); var email = ""; for (var i = 1; i < cuentasData.length; i++) { if (String(cuentasData[i][3]) === String(d.numCuenta)) { email = cuentasData[i][0]; break; } } if (!email) { return ContentService .createTextOutput(JSON.stringify({ok:false, error:"Cuenta "+d.numCuenta+" no encontrada"})) .setMimeType(ContentService.MimeType.JSON); } // 2. Buscar primera fila en COMANDES donde Product+Color+Size = Available var comandesSheet = ss.getSheetByName("COMANDES"); var comandesData = comandesSheet.getDataRange().getValues(); // Columnas: A=Order, B=Product, C=Size, D=Color, E=For, F=Status, G=Sold, H=Date var rowIndex = -1; for (var j = 1; j < comandesData.length; j++) { var prod = String(comandesData[j][1]).trim().toLowerCase(); var size = String(comandesData[j][2]).trim().toLowerCase(); var color = String(comandesData[j][3]).trim().toLowerCase(); var stat = String(comandesData[j][5]).trim().toLowerCase(); if ( prod === d.product.trim().toLowerCase() && size === String(d.size).trim().toLowerCase() && color === d.color.trim().toLowerCase() && stat === "available" ) { rowIndex = j + 1; // +1 porque getValues es 0-indexed pero Sheets es 1-indexed break; } } if (rowIndex === -1) { return ContentService .createTextOutput(JSON.stringify({ok:false, error:"No hay stock disponible de "+d.product+" "+d.color+" "+d.size})) .setMimeType(ContentService.MimeType.JSON); } // 3. Actualizar la fila encontrada comandesSheet.getRange(rowIndex, 5).setValue(email); // Col E: For comandesSheet.getRange(rowIndex, 6).setValue("Sold"); // Col F: Status comandesSheet.getRange(rowIndex, 7).setValue(d.price); // Col G: Sold (precio) comandesSheet.getRange(rowIndex, 9).setValue(new Date()); // Col I: Date (columna H es Bought) return ContentService .createTextOutput(JSON.stringify({ ok: true, row: rowIndex, email: email, product: d.product, color: d.color, size: d.size, price: d.price })) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { return ContentService .createTextOutput(JSON.stringify({status:"ok"})) .setMimeType(ContentService.MimeType.JSON); }
